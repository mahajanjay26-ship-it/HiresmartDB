### Find applications for each job
Select j.job_id,j.title,
count(a.application_id) as Total_Applications
from jobs j left join applications a 
on j.job_id=a.job_id
group by j.job_id,j.title
order by total_applications desc;


### Find total selected candidates per job.
select j.job_id,j.title,
count(a.application_id) as Total_Selected
from jobs j 
join applications a 
on j.job_id=a.job_id
where a.status='Selected'
group by j.job_id,j.title;


### Find all candidates whose experience is greater than 2 years.

### Find all jobs where maximum salary is greater than 500000.

### Find candidate name, job title, and assessment score only where score > 70.
desc candidates;
desc jobs;
desc applications;
desc assessments;

select c.name as candidate_name,j.title as job_title,
ass.score
from candidates c join applications a 
on c.candidate_id=a.candidate_id
join jobs j
on a.job_id=j.job_id 
join assessments ass
on a.application_id=ass.application_id
where ass.score > 70;

### Find total number of applications per company only for companies in 'Mumbai'
select c.company_name,
count(a.application_id) as total_applications
from companies c 
join jobs j
on c.company_id=j.company_id
join applications a 
on j.job_id=a.job_id
where c.location = 'Mumbai'
group by c.company_name;





select c.company_name,avg(ass.score) as average_score
from companies c join jobs j 
on c.company_id=j.company_id
join applications a
on j.job_id=a.job_id
join assessments ass
on a.application_id=ass.application_id
group by c.company_name;


### Top 3 highest scoring candidates
select c.name as candidate_name,max (ass.score) as highest_score
from candidates c join applications a
on c.candidate_id=a.candidate_id
join assessments ass
on a.application_id=ass.application_id
group by c.name
order by highest_score desc 
limit 3;

###  Recruiter-wise Total Hires
desc recruiters;
desc jobs;

select r.name as recruiter_name,
count(a.application_id) as total_hired
from recruiters r join jobs j 
on r.company_id=j.company_id
join applications a 
on a.job_id=j.job_id
join assessments ass
on a.application_id=ass.application_id
group by r.name
order by total_hired desc;



### Skill Demand Frequency
SELECT 
    s.skill_name,
    COUNT(cs.candidate_id) AS total_candidates
FROM skills s
JOIN candidate_skills cs 
    ON s.skill_id = cs.skill_id
GROUP BY s.skill_name
ORDER BY total_candidates DESC;
